function draw() { ctx.clearRect(0, 0, canvas.width, canvas.height); stars.forEach(s => { let speed = s.speed * warp; let alpha = 1; let lengthFactor = 3; // 🔥 ZONA SUPERIOR (texto) → suave, no desaparece if (s.y < canvas.height * 0.25) { alpha = 0.25; speed *= 0.4; lengthFactor = 1; } // 🔥 ZONA MEDIA (nave) → más sutil else if (s.y < canvas.height * 0.6) { alpha = 0.5; lengthFactor = 2; } // 🔥 ABAJO → máxima velocidad else { alpha = 1; lengthFactor = 4; } if (s.depth > 0.5) { const length = speed * lengthFactor; ctx.beginPath(); ctx.moveTo(s.x, s.y); ctx.lineTo(s.x + length, s.y); ctx.strokeStyle = `rgba(255,255,255,${alpha})`; ctx.lineWidth = s.depth * 2; ctx.stroke(); } else { ctx.beginPath(); ctx.arc(s.x, s.y, 1, 0, Math.PI * 2); ctx.fillStyle = `rgba(255,255,255,${alpha})`; ctx.fill(); } s.x -= speed; if (s.x < 0) { s.x = canvas.width; s.y = Math.random() * canvas.height; } }); requestAnimationFrame(draw); }
